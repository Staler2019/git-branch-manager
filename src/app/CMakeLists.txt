# App icon: macOS reads MACOSX_BUNDLE_ICON_FILE (set below) for the
# Finder/dock icon, which requires the referenced .icns to be a build
# *source* under Contents/Resources -- MACOSX_PACKAGE_LOCATION is what makes
# qt_add_executable's underlying add_executable() copy it there. Windows has
# no bundle equivalent; embedding app-icon.ico via a .rc resource script
# (added as a source too) is what puts it on the .exe itself. Neither
# artefact is rasterised at build time -- both are committed, generated by
# scripts/gen-app-icon.sh from resources/branding/app-icon.svg.
if(APPLE)
    target_sources(git-branch-manager PRIVATE "${CMAKE_SOURCE_DIR}/resources/branding/app-icon.icns")
    set_source_files_properties(
        "${CMAKE_SOURCE_DIR}/resources/branding/app-icon.icns"
        PROPERTIES MACOSX_PACKAGE_LOCATION "Resources")
elseif(WIN32)
    target_sources(git-branch-manager PRIVATE "${CMAKE_SOURCE_DIR}/resources/branding/app-icon.rc")
endif()

# qt_add_executable() enables AUTOMOC automatically but NOT AUTORCC (see the
# identical comment in tests/CMakeLists.txt, which discovered this the same
# way) -- without this, resources/app.qrc sits in the source list as a inert,
# unprocessed file. The practical effect was silent: ThemeManager::apply()'s
# QFile::open(":/qss/app.qss") simply failed every time (verified: it returns
# false, error 5 "file not found"), so qApp->setStyleSheet() was never called
# in any build of the actual application -- only the two test binaries ever
# actually loaded the stylesheet, because they set this explicitly. Every
# color came from the QPalette (set directly from token values, no qrc
# needed) and every hand-painted delegate (GraphColumnDelegate, RefRowDelegate,
# CommitRowDelegate) reads ThemeManager::color() directly -- so the app was
# never blank, just running with zero of its own QSS: default Fusion metrics,
# no custom fonts (the bundled Inter/JetBrains Mono .ttf never registered
# either, since QFontDatabase::addApplicationFont(":/fonts/...") hit the same
# missing-resource failure), and every "QSS" fix in the UI redesign plan had
# had no effect until this line existed.
set_target_properties(git-branch-manager PROPERTIES AUTORCC ON)

target_include_directories(git-branch-manager PRIVATE
    "${CMAKE_CURRENT_SOURCE_DIR}/.."
    "${CMAKE_CURRENT_BINARY_DIR}/generated")

target_link_libraries(git-branch-manager
    PRIVATE gbm_core Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Svg gbm_warnings gbm_sanitizers)

set_target_properties(git-branch-manager PROPERTIES
    WIN32_EXECUTABLE ON
    MACOSX_BUNDLE ON
    MACOSX_BUNDLE_BUNDLE_NAME "git-branch-manager"
    MACOSX_BUNDLE_GUI_IDENTIFIER "dev.gitbranchmanager.app"
    MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}"
    MACOSX_BUNDLE_ICON_FILE "app-icon.icns")

# LGPL compliance: Qt must stay dynamically linked so a user can replace it. A
# static Qt would put obligations on us we cannot meet for a distributed binary.
get_target_property(_qt_core_type Qt6::Core TYPE)
if(_qt_core_type STREQUAL "STATIC_LIBRARY")
    message(FATAL_ERROR
        "Qt appears to be a static build. Distributing a statically linked Qt is not "
        "permitted under the LGPL without providing relinking materials; use a shared Qt.")
endif()

install(TARGETS git-branch-manager
    BUNDLE DESTINATION .
    RUNTIME DESTINATION bin)

    "${CMAKE_SOURCE_DIR}/THIRD_PARTY_NOTICES.md"
    DESTINATION .)

# --- runtime deployment ----------------------------------------------------
# Each platform's Qt deployment tool copies the Qt shared libraries next to the
# binary, which is both what makes the app runnable and what satisfies the LGPL's
# dynamic-linking requirement.
if(WIN32)
    find_program(GBM_WINDEPLOYQT windeployqt HINTS "${QT6_INSTALL_PREFIX}/bin"
                 "${Qt6_DIR}/../../../bin")
    if(GBM_WINDEPLOYQT)
        install(CODE "
            execute_process(COMMAND \"${GBM_WINDEPLOYQT}\"
                --release --no-translations --no-system-d3d-compiler --no-opengl-sw
                \"\${CMAKE_INSTALL_PREFIX}/bin/git-branch-manager.exe\")")
    else()
        message(WARNING "windeployqt not found; the installed build will not be self-contained")
    endif()
elseif(APPLE)
    find_program(GBM_MACDEPLOYQT macdeployqt HINTS "${QT6_INSTALL_PREFIX}/bin"
                 "${Qt6_DIR}/../../../bin")
    if(GBM_MACDEPLOYQT)
        install(CODE "
            execute_process(COMMAND \"${GBM_MACDEPLOYQT}\"
                \"\${CMAKE_INSTALL_PREFIX}/git-branch-manager.app\")")
    else()
        message(WARNING "macdeployqt not found; the bundle will not be self-contained")
    endif()
endif()
